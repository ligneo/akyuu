cmake_minimum_required(VERSION 3.21)
project(AkyuuRelease LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core)
add_executable(akyuu-release main.cpp common.cpp archive.cpp release.cpp
	sources.cpp source_tests.cpp common_tests.cpp linux.cpp windows.cpp)
target_link_libraries(akyuu-release PRIVATE Qt6::Core)
if(WIN32)
	target_link_libraries(akyuu-release PRIVATE wintrust crypt32 shell32 ole32 advapi32)
endif()
if(MSVC)
	target_compile_options(akyuu-release PRIVATE /W4 /permissive-)
else()
	target_compile_options(akyuu-release PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(akyuu-release PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS akyuu-release RUNTIME DESTINATION bin)
enable_testing()
add_test(NAME source-companion COMMAND akyuu-release source-tests)
add_test(NAME release-utility COMMAND akyuu-release common-tests)
